% Vaciamos encabezados y pies de pagina
\makeatletter
\patchcmd{\chapter}
  {\thispagestyle{plain}}
  {\thispagestyle{empty}}
  {}{}
\makeatother

% Caja para el número del capítulo
\newsavebox{\felinechapternumber}

% Marcador: "CAPÍTULO" vertical + número
\newcommand{\felinechaptermarker}{%
    \sbox{\felinechapternumber}{%
        \begingroup
        \setlength{\fboxsep}{4pt}%
        \colorbox{rojoCIMAT}{%
            \resizebox{!}{4.9cm}{%
                \normalfont
                \bfseries
                \color{white}%
                \thechapter
            }%
        }%
        \endgroup
    }%
    \makebox[\textwidth][r]{%
        {%
            \normalfont
            \rmfamily
            \mdseries
            \color{black}%
            \rotatebox{90}{%
                \resizebox{%
                    \heightof{\usebox{\felinechapternumber}}%
                    +\depthof{\usebox{\felinechapternumber}}%
                }{!}{%
                    \scshape\so{CAPÍTULO}%
                }%
            }%
        }%
        \hspace{0.35em}%
        \raisebox{\depthof{\usebox{\felinechapternumber}}}{%
            \usebox{\felinechapternumber}%
        }%
    }%
}
