\newcommand{\myref}[1]{%
  \cref{#1}%
  \begingroup
  \edef\nombrereferencia{\getrefbykeydefault{#1}{name}{}}%
  \ifx\nombrereferencia\empty\else~(\nameref{#1})\fi
  \endgroup
}

  \begingroup
  \crefname{teo}{teo.}{teos.}%

  \crefname{ejercicio}{ejerc.}{ejercs.}%
  \crefname{section}{sec.}{secs.}%
  \crefname{subsection}{subsec.}{subsecs.}%
  \cref{#1}%
  \endgroup
